# IraqJobs V2 Full 🇮🇶

منصة وظائف عراقية متعددة الأدوار مبنية كـ Monorepo:
- `mobile`: تطبيق Expo + React Native + TypeScript
- `backend`: API + PostgreSQL + Prisma + Socket.IO
- `docs`: توثيق التشغيل والبنية
- `.github`: GitHub Actions

## المزايا
- الباحث عن عمل وصاحب العمل والإدارة
- تسجيل/دخول JWT
- صلاحيات ديناميكية Role/Permission
- وظائف + بحث + فلترة حسب المحافظة
- تقديم على الوظائف ومتابعة الحالة
- شركات
- ملف شخصي وسيرة ذاتية
- مفضلة
- محادثات مرتبطة بالحساب
- Socket.IO للرسائل الفورية
- إشعارات داخلية
- بلاغات
- إدارة المحافظات والأقضية
- لوحة إدارة API
- حماية Helmet + CORS + rate limit + Zod + bcrypt
- PostgreSQL + Prisma
- Docker Compose
- GitHub CI

## التشغيل المحلي

### Backend
```bash
cd backend
cp .env.example .env
npm install
npx prisma generate
npx prisma migrate dev --name init
npm run seed
npm run dev
```

### Mobile
```bash
cd mobile
npm install
npx expo start
```

> على جهاز Android حقيقي، لا تستخدم `localhost` للـAPI؛ ضع IP الكمبيوتر في `mobile/src/config/env.ts`.

import { PrismaClient, UserType } from "@prisma/client";
import bcrypt from "bcryptjs";

const prisma = new PrismaClient();

const provinces = [
  "بغداد","البصرة","نينوى","أربيل","النجف","كربلاء","بابل","ذي قار",
  "ميسان","واسط","ديالى","الأنبار","صلاح الدين","كركوك","الديوانية",
  "المثنى","دهوك","السليمانية","حلبجة"
];

const permissionKeys = [
  "jobs.read","jobs.create","jobs.update","jobs.delete",
  "applications.read","applications.update",
  "users.read","users.update","users.delete",
  "chat.use","notifications.read","reports.create","reports.manage",
  "roles.manage","companies.create","companies.update"
];

async function main() {
  for (const nameAr of provinces) {
    await prisma.province.upsert({where:{nameAr},update:{},create:{nameAr}});
  }

  for (const key of permissionKeys) {
    await prisma.permission.upsert({where:{key},update:{},create:{key}});
  }

  const roleNames = ["ADMIN","EMPLOYER","SEEKER"];
  const roles: Record<string,string> = {};
  for (const name of roleNames) {
    const role = await prisma.role.upsert({where:{name},update:{},create:{name}});
    roles[name] = role.id;
  }

  const adminKeys = permissionKeys;
  const employerKeys = ["jobs.read","jobs.create","jobs.update","applications.read","applications.update","chat.use","notifications.read","reports.create","companies.create","companies.update"];
  const seekerKeys = ["jobs.read","applications.read","chat.use","notifications.read","reports.create"];

  for (const [roleName, keys] of [["ADMIN",adminKeys],["EMPLOYER",employerKeys],["SEEKER",seekerKeys]] as const) {
    for (const key of keys) {
      const permission = await prisma.permission.findUniqueOrThrow({where:{key}});
      await prisma.rolePermission.upsert({
        where:{roleId_permissionId:{roleId:roles[roleName],permissionId:permission.id}},
        update:{},create:{roleId:roles[roleName],permissionId:permission.id}
      });
    }
  }

  const accounts = [
    {email:"admin@iraqjobs.local",name:"مدير النظام",type:UserType.ADMIN,password:"Admin123!",role:"ADMIN"},
    {email:"employer@iraqjobs.local",name:"صاحب عمل تجريبي",type:UserType.EMPLOYER,password:"Employer123!",role:"EMPLOYER"},
    {email:"user@iraqjobs.local",name:"باحث عن عمل",type:UserType.SEEKER,password:"User123!",role:"SEEKER"}
  ];

  const created:any = {};
  for (const a of accounts) {
    const user = await prisma.user.upsert({
      where:{email:a.email},update:{name:a.name,type:a.type},
      create:{email:a.email,name:a.name,type:a.type,passwordHash:await bcrypt.hash(a.password,12)}
    });
    created[a.role] = user;
    await prisma.userRole.upsert({
      where:{userId_roleId:{userId:user.id,roleId:roles[a.role]}},
      update:{},create:{userId:user.id,roleId:roles[a.role]}
    });
  }

  const baghdad = await prisma.province.findUniqueOrThrow({where:{nameAr:"بغداد"}});
  const company = await prisma.company.upsert({
    where:{ownerId:created.EMPLOYER.id},update:{},
    create:{ownerId:created.EMPLOYER.id,name:"شركة العراق للتقنية",description:"شركة تجريبية لمنصة وظائف العراق",provinceId:baghdad.id}
  });

  await prisma.job.upsert({
    where:{id:"demo-job-1"},update:{},
    create:{
      id:"demo-job-1",title:"موظف مبيعات",description:"فرصة عمل تجريبية داخل بغداد. يشترط التواصل الجيد والالتزام.",
      salaryMin:500000,salaryMax:800000,type:"FULL_TIME",companyName:company.name,
      provinceId:baghdad.id,employerId:created.EMPLOYER.id,companyId:company.id
    }
  });

  console.log("Seed complete.");
}

main().catch(console.error).finally(()=>prisma.$disconnect());

import jwt from "jsonwebtoken";
import bcrypt from "bcryptjs";
import { env } from "./config/env";

export type TokenPayload = {sub:string; type:string};

export function signToken(payload:TokenPayload) {
  return jwt.sign(payload, env.JWT_SECRET, {expiresIn:env.JWT_EXPIRES_IN as jwt.SignOptions["expiresIn"]});
}
export function verifyToken(token:string) {
  return jwt.verify(token, env.JWT_SECRET) as TokenPayload;
}
export const hashPassword=(value:string)=>bcrypt.hash(value,12);
export const comparePassword=(value:string,hash:string)=>bcrypt.compare(value,hash);

import {Request,Response,NextFunction} from "express";
import {verifyToken} from "./auth";
import {prisma} from "./db";

declare global {
  namespace Express {
    interface Request {
      user?: {id:string; type:string; permissions:Set<string>};
    }
  }
}

export async function requireAuth(req:Request,res:Response,next:NextFunction) {
  try {
    const header=req.headers.authorization;
    if(!header?.startsWith("Bearer ")) return res.status(401).json({message:"غير مصرح"});
    const payload=verifyToken(header.slice(7));
    const user=await prisma.user.findUnique({
      where:{id:payload.sub},
      include:{roles:{include:{role:{include:{permissions:{include:{permission:true}}}}}}}
    });
    if(!user||!user.isActive) return res.status(401).json({message:"الحساب غير فعال"});
    const permissions=new Set(user.roles.flatMap(r=>r.role.permissions.map(x=>x.permission.key)));
    req.user={id:user.id,type:user.type,permissions};
    next();
  } catch { res.status(401).json({message:"رمز الدخول غير صالح"}); }
}

export function permission(key:string) {
  return (req:Request,res:Response,next:NextFunction)=>{
    if(!req.user?.permissions.has(key)) return res.status(403).json({message:"لا تملك الصلاحية"});
    next();
  };
}

export function adminOnly(req:Request,res:Response,next:NextFunction) {
  if(req.user?.type!=="ADMIN") return res.status(403).json({message:"للمدير فقط"});
  next();
}

    phone:z.string().max(30).optional(),type:z.enum(["SEEKER","EMPLOYER"]).default("SEEKER")
  }).parse(req.body);
  if(await prisma.user.findUnique({where:{email:data.email}})) return res.status(409).json({message:"البريد مستخدم"});
  const user=await prisma.user.create({data:{
    email:data.email,name:data.name,phone:data.phone,type:data.type,passwordHash:await hashPassword(data.password)
  }});
  const role=await prisma.role.findUnique({where:{name:data.type}});
  if(role) await prisma.userRole.create({data:{userId:user.id,roleId:role.id}});
  res.status(201).json({token:signToken({sub:user.id,type:user.type}),user:{id:user.id,name:user.name,email:user.email,type:user.type}});
});

import {Router} from "express";
import {z} from "zod";
import {prisma} from "../db";
import {permission,requireAuth} from "../middleware";

const router=Router();
const createSchema=z.object({
  title:z.string().min(2).max(150),description:z.string().min(10),
  salaryMin:z.number().int().nonnegative().optional(),salaryMax:z.number().int().nonnegative().optional(),
  type:z.enum(["FULL_TIME","PART_TIME","CONTRACT","REMOTE","FREELANCE"]),
  companyName:z.string().min(2),address:z.string().max(300).optional(),
  provinceId:z.string(),districtId:z.string().optional(),companyId:z.string().optional()
});

router.get("/",async(req,res)=>{
  const q=typeof req.query.q==="string"?req.query.q:undefined;
  const provinceId=typeof req.query.provinceId==="string"?req.query.provinceId:undefined;
  const type=typeof req.query.type==="string"?req.query.type:undefined;
  const jobs=await prisma.job.findMany({
    where:{isActive:true,...(provinceId?{provinceId}:{}),...(type?{type:type as any}:{}),
      ...(q?{OR:[{title:{contains:q,mode:"insensitive"}},{description:{contains:q,mode:"insensitive"}},{companyName:{contains:q,mode:"insensitive"}}]}:{})},
    include:{province:true,district:true,company:true},orderBy:{createdAt:"desc"},take:100
  });
  res.json(jobs);
});

router.get("/:id",async(req,res)=>{
  const job=await prisma.job.findUnique({where:{id:req.params.id},include:{province:true,district:true,company:true,employer:{select:{id:true,name:true}}}});
  if(!job)return res.status(404).json({message:"الوظيفة غير موجودة"});
  res.json(job);
});

router.post("/",requireAuth,permission("jobs.create"),async(req,res)=>{
  const data=createSchema.parse(req.body);
  const job=await prisma.job.create({data:{...data,employerId:req.user!.id}});
  res.status(201).json(job);
});

router.patch("/:id",requireAuth,permission("jobs.update"),async(req,res)=>{
  const data=createSchema.partial().parse(req.body);
  const job=await prisma.job.findUnique({where:{id:req.params.id}});
  if(!job)return res.status(404).json({message:"الوظيفة غير موجودة"});
  if(req.user!.type!=="ADMIN"&&job.employerId!==req.user!.id)return res.status(403).json({message:"غير مسموح"});
  res.json(await prisma.job.update({where:{id:job.id},data}));
});

router.delete("/:id",requireAuth,permission("jobs.delete"),async(req,res)=>{
  const job=await prisma.job.findUnique({where:{id:req.params.id}});
  if(!job)return res.status(404).json({message:"غير موجودة"});
  if(req.user!.type!=="ADMIN"&&job.employerId!==req.user!.id)return res.status(403).json({message:"غير مسموح"});
  await prisma.job.update({where:{id:job.id},data:{isActive:false}});
  res.status(204).send();
});

import rateLimit from "express-rate-limit";
import {createServer} from "http";
import {Server} from "socket.io";
import {env} from "./config/env";
import {prisma} from "./db";
import {verifyToken} from "./auth";
import {errorHandler} from "./error";
import auth from "./routes/auth";
import jobs from "./routes/jobs";
import applications from "./routes/applications";
import locations from "./routes/locations";
import profile from "./routes/profile";
import companies from "./routes/companies";
import favorites from "./routes/favorites";
import notifications from "./routes/notifications";
import chat from "./routes/chat";
import reports from "./routes/reports";

const app=express();
const server=createServer(app);
const io=new Server(server,{cors:{origin:env.CORS_ORIGIN==="*"?"*":env.CORS_ORIGIN}});
app.use(helmet());
app.use(cors({origin:env.CORS_ORIGIN==="*"?"*":env.CORS_ORIGIN}));
app.use(express.json({limit:"2mb"}));
app.use(morgan("combined"));
app.use(rateLimit({windowMs:60_000,max:180}));

app.get("/health",(_req,res)=>res.json({ok:true,service:"iraqjobs-api",version:"2.0.0"}));
app.use("/api/auth",auth);
app.use("/api/jobs",jobs);
app.use("/api/applications",applications);
app.use("/api/locations",locations);
app.use("/api/profile",profile);
app.use("/api/companies",companies);
app.use("/api/favorites",favorites);
app.use("/api/notifications",notifications);
app.use("/api/chat",chat);
app.use("/api/reports",reports);
app.use("/api/admin",admin);
app.use(errorHandler);

io.use((socket,next)=>{
  try {
    const token=socket.handshake.auth?.token;
    if(!token)return next(new Error("unauthorized"));
    const payload=verifyToken(token);
    socket.data.userId=payload.sub;
    next();
  } catch { next(new Error("unauthorized")); }
});

io.on("connection",(socket)=>{
  socket.on("conversation:join",async(conversationId:string)=>{
    const member=await prisma.conversationMember.findUnique({where:{conversationId_userId:{conversationId,userId:socket.data.userId}}});
    if(member)socket.join(`conversation:${conversationId}`);
  });
  socket.on("message:send",async(data:{conversationId:string;body:string})=>{
    if(!data?.body?.trim())return;
    const member=await prisma.conversationMember.findUnique({where:{conversationId_userId:{conversationId:data.conversationId,userId:socket.data.userId}}});
    if(!member)return;
    const msg=await prisma.message.create({data:{conversationId:data.conversationId,senderId:socket.data.userId,body:data.body.trim()}});
    io.to(`conversation:${data.conversationId}`).emit("message:new",msg);
  });
});

server.listen(env.PORT,()=>console.log(`IraqJobs API listening on ${env.PORT}`));
process.on("SIGINT",async()=>{await prisma.$disconnect();process.exit(0)});
process.on("SIGTERM",async()=>{await prisma.$disconnect();process.exit(0)});

import {API_BASE_URL} from "../config/env";
import {useAuth} from "../store/auth";

export async function api<T>(path:string,options:RequestInit={}):Promise<T>{
 const token=useAuth.getState().token;
 const headers=new Headers(options.headers);
 headers.set("Content-Type","application/json");
 if(token)headers.set("Authorization",`Bearer ${token}`);
 const res=await fetch(`${API_BASE_URL}${path}`,{...options,headers});
 const data=await res.json().catch(()=>null);
 if(!res.ok)throw new Error(data?.message||"تعذر الاتصال بالخادم");
 return data as T;
}

import {create} from "zustand";
import * as SecureStore from "expo-secure-store";
import type {User} from "../types";

type State={
 token:string|null;user:User|null;loading:boolean;
 setSession:(token:string,user:User)=>Promise<void>;
 logout:()=>Promise<void>;restore:()=>Promise<void>;
};
export const useAuth=create<State>((set)=>({
 token:null,user:null,loading:true,
 setSession:async(token,user)=>{await SecureStore.setItemAsync("token",token);await SecureStore.setItemAsync("user",JSON.stringify(user));set({token,user,loading:false});},
 logout:async()=>{await SecureStore.deleteItemAsync("token");await SecureStore.deleteItemAsync("user");set({token:null,user:null,loading:false});},
 restore:async()=>{const [token,user]=await Promise.all([SecureStore.getItemAsync("token"),SecureStore.getItemAsync("user")]);set({token,user:user?JSON.parse(user):null,loading:false});}
}));
